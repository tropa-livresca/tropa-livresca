# Documentação do Modelo de Dados (Supabase/PostgreSQL)

Este documento contém as diretrizes e instruções técnicas sobre o mapeamento estrutural das tabelas do banco de dados do projeto. Ele serve como documentação conceitual e guia de implantação para a arquitetura de persistência.

---

## Instruções para Execução e Implantação

Se você precisar recriar estas tabelas em um banco de dados limpo ou em um novo ambiente do Supabase, siga rigorosamente as diretrizes abaixo:

1. **Geração Automática de IDs (Identity):**
   - As sequências numéricas utilizam a cláusula nativa e padronizada `GENERATED BY DEFAULT AS IDENTITY`, eliminando dependências manuais de objetos sequenciais externos.
   - A tabela `users_profile` adota o tipo `uuid` integrado ao ecossistema do Supabase Auth para controle centralizado de chaves primárias.

2. **Chaves Estrangeiras Pendentes:**
   - Este documento foca estritamente na estrutura estrutural de criação das tabelas. Os comandos de amarração de chaves estrangeiras (`ALTER TABLE ... ADD CONSTRAINT`) devem ser executados apenas **após** a criação bem-sucedida de todas as entidades listadas na arquitetura.

3. **Esquema de Destino:**
   - Todas as entidades estão mapeadas explicitamente para o esquema `public`.

---

## Estrutura de Criação das Tabelas (DDL)

```sql
-- ==========================================
-- CRIAÇÃO DA TABELA: users_profile
-- ==========================================

CREATE TABLE public.users_profile (
  id uuid not null,
  nome text not null,
  telefone character varying(15) not null,
  imagem text null,
  descricao text null,
  redes_sociais jsonb null,
  is_admin boolean not null default false,
  senha_adm text null default extensions.pgp_sym_encrypt (
    'Senh@Adm123!'::text,
    '225419eb-fd25-4430-8103-9591d2e4361d'::text
  ),
  primeiro_acesso boolean not null default true,
  email text null,
  is_master boolean null default false,
  constraint users_profile_pkey primary key (id),
  constraint users_profile_id_fkey foreign KEY (id) references auth.users (id) on delete CASCADE
) TABLESPACE pg_default;

create table public.avaliacoes (
  id bigint generated by default as identity not null,
  data_criacao timestamp with time zone not null default now(),
  fk_livros_id bigint null,
  fk_users_profile_id uuid null,
  qtd_estrelas smallint null,
  constraint avaliacoes_pkey primary key (id),
  constraint avaliacoes_fk_livros_id_fkey foreign KEY (fk_livros_id) references livros (id),
  constraint avaliacoes_fk_users_profile_id_fkey foreign KEY (fk_users_profile_id) references users_profile (id)
) TABLESPACE pg_default;

-- ==========================================
-- CRIAÇÃO DA TABELA: livros
-- ==========================================
CREATE TABLE public.livros (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    direitos_de_publicacao boolean NOT NULL,
    conteudo_por_IA boolean NOT NULL,
    imagens_explicitas boolean NOT NULL,
    numero_edicao smallint,
    data_de_publicacao date NOT NULL,
    preco_digital numeric NOT NULL,
    preco_fisico numeric NOT NULL,
    titulo text NOT NULL,
    subtitulo text NOT NULL,
    descricao text NOT NULL,
    autor_nome text NOT NULL,
    autor_sobrenome text NOT NULL,
    fk_user_profile_id uuid NOT NULL,
    ativo boolean NOT NULL DEFAULT true,
    estado character varying NOT NULL DEFAULT 'rascunho'::character varying,
    idioma text NOT NULL,
    manuscrito text,
    colaboradores jsonb DEFAULT '[]'::jsonb,
    total_paginas numeric NOT NULL,
    capa jsonb DEFAULT '{}'::jsonb,
    categoria text NOT NULL,
    CONSTRAINT livros_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: comentarios
-- ==========================================
CREATE TABLE public.comentarios (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    data timestamp with time zone DEFAULT now() NOT NULL,
    texto text NOT NULL,
    fk_user_profile_id uuid NOT NULL,
    fk_livros_id bigint,
    CONSTRAINT comentarios_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: enderecos
-- ==========================================
CREATE TABLE public.enderecos (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    estado character varying NOT NULL,
    bairro text NOT NULL,
    rua text NOT NULL,
    num integer NOT NULL,
    cep character varying NOT NULL,
    fk_user_profile_id uuid NOT NULL,
    cidade text NOT NULL,
    complemento text,
    CONSTRAINT enderecos_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: met_pagamentos
-- ==========================================
CREATE TABLE public.met_pagamentos (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    nome text NOT NULL,
    CONSTRAINT met_pagamentos_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: usuarios_cartoes
-- ==========================================
CREATE TABLE public.usuarios_cartoes (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    token_pagamento text,
    final_cartao character varying NOT NULL,
    bandeira_varchar character varying NOT NULL,
    nome_no_cartao text NOT NULL,
    data_expiracao character varying NOT NULL,
    fk_user_profile_id uuid NOT NULL,
    CONSTRAINT usuarios_cartoes_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: vendas
-- ==========================================
CREATE TABLE public.vendas (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    data timestamp with time zone DEFAULT now() NOT NULL,
    total numeric NOT NULL,
    fk_met_pagamentos_id bigint NOT NULL,
    transacao_id text NOT NULL,
    status_pagamento text NOT NULL,
    fk_user_profile_id uuid NOT NULL,
    endereco_entrega jsonb NOT NULL,
    CONSTRAINT vendas_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: itens_venda
-- ==========================================
CREATE TABLE public.itens_venda (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    qtd bigint NOT NULL,
    subtotal numeric NOT NULL,
    formato text NOT NULL,
    fk_livros_itens_id bigint NOT NULL,
    fk_vendas_id bigint NOT NULL,
    CONSTRAINT itens_venda_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: revisoes
-- ==========================================
CREATE TABLE public.revisoes (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    nome text NOT NULL,
    arquivo text NOT NULL,
    data timestamp with time zone DEFAULT now() NOT NULL,
    apontamento text,
    fk_user_profile_id uuid NOT NULL,
    fk_livro_id bigint NOT NULL,
    CONSTRAINT revisoes_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: notificacoes
-- ==========================================
CREATE TABLE public.notificacoes (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    titulo text NOT NULL,
    texto text NOT NULL,
    tipo text NOT NULL,
    data timestamp with time zone DEFAULT now() NOT NULL,
    ADM boolean DEFAULT false NOT NULL,
    fk_user_profile_id uuid,
    geral boolean DEFAULT true NOT NULL,
    metadados jsonb DEFAULT '{}'::jsonb,
    CONSTRAINT notificacoes_pkey PRIMARY KEY (id)
);

-- ==========================================
-- CRIAÇÃO DA TABELA: notificacoes_lidas
-- ==========================================
CREATE TABLE public.notificacoes_lidas (
    id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    fk_notificacao_id bigint NOT NULL,
    fk_user_profile_id uuid NOT NULL,
    data_leitura timestamp with time zone DEFAULT now() NOT NULL,
    CONSTRAINT notificacoes_lidas_pkey PRIMARY KEY (id),
    CONSTRAINT unique_usuario_leitura_notificacao UNIQUE (fk_notificacao_id, fk_user_profile_id)
);

```

## Relacionamentos e Restrições (Chaves Estrangeiras)

As diretivas abaixo realizam as amarrações lógicas entre as entidades do ecossistema. Certifique-se de executar este bloco **apenas após** a criação de todas as tabelas listadas anteriormente.

```sql
-- ==========================================
-- CONSTRANGIMENTOS: livros
-- ==========================================
ALTER TABLE public.livros
    ADD CONSTRAINT fk_livros_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE CASCADE;

-- ==========================================
-- CONSTRANGIMENTOS: comentarios
-- ==========================================
ALTER TABLE public.comentarios
    ADD CONSTRAINT fk_comentarios_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE CASCADE,
    ADD CONSTRAINT fk_comentarios_livro FOREIGN KEY (fk_livros_id) REFERENCES public.livros(id) ON DELETE CASCADE,

-- ==========================================
-- CONSTRANGIMENTOS: enderecos
-- ==========================================
ALTER TABLE public.enderecos
    ADD CONSTRAINT fk_enderecos_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE CASCADE;

-- ==========================================
-- CONSTRANGIMENTOS: usuarios_cartoes
-- ==========================================
ALTER TABLE public.usuarios_cartoes
    ADD CONSTRAINT fk_cartoes_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE CASCADE;

-- ==========================================
-- CONSTRANGIMENTOS: vendas
-- ==========================================
ALTER TABLE public.vendas
    ADD CONSTRAINT fk_vendas_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE RESTRICT,
    ADD CONSTRAINT fk_vendas_met_pagamento FOREIGN KEY (fk_met_pagamentos_id) REFERENCES public.met_pagamentos(id) ON DELETE RESTRICT;

-- ==========================================
-- CONSTRANGIMENTOS: itens_venda
-- ==========================================
ALTER TABLE public.itens_venda
    ADD CONSTRAINT fk_itens_venda_pai FOREIGN KEY (fk_vendas_id) REFERENCES public.vendas(id) ON DELETE CASCADE,
    ADD CONSTRAINT fk_itens_venda_livro FOREIGN KEY (fk_livros_itens_id) REFERENCES public.livros(id) ON DELETE RESTRICT;

-- ==========================================
-- CONSTRANGIMENTOS: revisoes
-- ==========================================
ALTER TABLE public.revisoes
    ADD CONSTRAINT fk_revisoes_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE CASCADE,
    ADD CONSTRAINT fk_revisoes_livro FOREIGN KEY (fk_livro_id) REFERENCES public.livros(id) ON DELETE CASCADE;

-- ==========================================
-- CONSTRANGIMENTOS: notificacoes
-- ==========================================
ALTER TABLE public.notificacoes
    ADD CONSTRAINT fk_notificacoes_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE CASCADE;

-- ==========================================
-- CONSTRANGIMENTOS: notificacoes_lidas
-- ==========================================
ALTER TABLE public.notificacoes_lidas
    ADD CONSTRAINT fk_lidas_notificacao FOREIGN KEY (fk_notificacao_id) REFERENCES public.notificacoes(id) ON DELETE CASCADE,
    ADD CONSTRAINT fk_lidas_user FOREIGN KEY (fk_user_profile_id) REFERENCES public.users_profile(id) ON DELETE CASCADE;
```
